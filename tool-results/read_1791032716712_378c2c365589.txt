   121→   121→function StatusIcon({ status }: { status: "ok" | "warn" | "fail" | string }) {
   122→   122→  if (status === "ok") return <CheckCircle2 size={15} className="shrink-0 text-emerald-400" />;
   123→   123→  if (status === "warn") return <AlertTriangle size={15} className="shrink-0 text-amber-400" />;
   124→   124→  return <XCircle size={15} className="shrink-0 text-red-400" />;
   125→   125→}
   126→   126→
   127→   127→function Pill({ children, tone = "neutral" }: { children: React.ReactNode; tone?: "neutral" | "good" | "warn" | "bad" | "accent" }) {
   128→   128→  const tones = {
   129→   129→    neutral: "bg-[var(--elevated)] text-[var(--fg-dim)]",
   130→   130→    good: "bg-emerald-500/15 text-emerald-400",
   131→   131→    warn: "bg-amber-500/15 text-amber-400",
   132→   132→    bad: "bg-red-500/15 text-red-400",
   133→   133→    accent: "bg-[var(--primary)]/15 text-[var(--primary)]",
   134→   134→  };
   135→   135→  return <span className={cn("rounded-[4px] px-2 py-0.5 text-[11px] font-bold", tones[tone])}>{children}</span>;
   136→   136→}

   151→   151→function timeAgo(iso?: string | number | null): string {
   152→   152→  if (!iso) return "never";
   153→   153→  try {
   154→   154→    return formatDistanceToNow(typeof iso === "number" ? new Date(iso) : new Date(iso), { addSuffix: true });
   155→   155→  } catch {
   156→   156→    return "unknown";
   157→   157→  }
   158→   158→}

   168→   168→  const [kindFilter, setKindFilter] = useState<string>("");

   170→   170→  const [busy, setBusy] = useState<string>("");

   175→   175→      const res = await fetch("/api/agent/status", { cache: "no-store" });

   177→   177→        const data = (await res.json()) as { ok: boolean } & AgentStatus;
   178→   178→        if (data.ok) setStatus(data);
   179→   179→      }
   180→   180→    } catch {
   181→   181→      /* transient */
   182→   182→    }
   183→   183→  }, []);
   184→   184→
   185→   185→  const loadFindings = useCallback(async () => {
   186→   186→    try {
   187→   187→      const qs = new URLSearchParams({ limit: "80" });
   188→   188→      if (kindFilter) qs.set("kind", kindFilter);
   189→   189→      const res = await fetch(`/api/agent/findings?${qs}`, { cache: "no-store" });
   190→   190→      if (res.ok) {
   191→   191→        const data = (await res.json()) as { ok: boolean; total: number; items: FindingRow[] };
   192→   192→        if (data.ok) {
   193→   193→          setFindings(data.items);
   194→   194→          setFindingsTotal(data.total);
   195→   195→        }
   196→   196→      }
   197→   197→    } catch {
   198→   198→      /* transient */
   199→   199→    }
   200→   200→  }, [kindFilter]);
   201→   201→
   202→   202→  const loadHistory = useCallback(async () => {
   203→   203→    try {
   204→   204→      const res = await fetch("/api/agent/health?history=60", { cache: "no-store" });
   205→   205→      if (res.ok) {
   206→   206→        const data = (await res.json()) as { ok: boolean; history: HistoryPoint[] };
   207→   207→        if (data.ok) setHistory(data.history);
   208→   208→      }
   209→   209→    } catch {
   210→   210→      /* transient */
   211→   211→    }
   212→   212→  }, []);
   213→   213→
   214→   214→  const loadRuns = useCallback(async () => {
   215→   215→    try {
   216→   216→      const res = await fetch("/api/agent/status", { cache: "no-store" });
   217→   217→      void res;
   218→   218→      // runs come from the status endpoint's last* fields; full history via runs query below
   219→   219→    } catch {
   220→   220→      /* transient */
   221→   221→    }
   222→   222→  }, []);
   223→   223→
   224→   224→  useEffect(() => {
   225→   225→    void loadStatus();
   226→   226→    void loadFindings();
   227→   227→    void loadHistory();

   229→   229→    return () => {
   230→   230→      if (pollRef.current) clearInterval(pollRef.current);
   231→   231→    };
   232→   232→  }, [loadStatus, loadFindings, loadHistory]);
   233→   233→
   234→   234→  useEffect(() => {
   235→   235→    void loadFindings();
   236→   236→  }, [kindFilter, loadFindings]);

   240→   240→    try {
   241→   241→      const res = await fetch("/api/agent/run", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ job }) });
   242→   242→      const data = (await res.json()) as { ok: boolean; started?: boolean; error?: string };
   243→   243→      if (data.ok) {
   244→   244→        toast(data.started ? `${job === "scan" ? "Library scan" : "Health check"} started` : "Job already running", { duration: 2000 });
   245→   245→        setTimeout(() => {
   246→   246→          void loadStatus();
   247→   247→          void loadFindings();
   248→   248→          void loadHistory();
   249→   249→        }, 1500);
   250→   250→      } else {
   251→   251→        toast.error(data.error ?? "Failed to start job");
   252→   252→      }
   253→   253→    } catch {
   254→   254→      toast.error("Agent unreachable");
   255→   255→    } finally {
   256→   256→      setBusy("");

   264→   264→        await loadStatus();

   266→   266→      } else {
   267→   267→        toast.error("Failed to save settings");

   279→   279→          {Array.from({ length: 4 }).map((_, i) => (

   294→   294→  return (
   295→   295→    <div className="px-8 pb-24 pt-8" data-testid="agent-view">
   296→   296→      {/* header */}
   297→   297→      <div className="mb-6 flex flex-wrap items-center gap-3">
   298→   298→        <div className="flex h-11 w-11 items-center justify-center rounded-[8px] bg-[var(--primary)]/15 text-[var(--primary)]">
   299→   299→          <Bot size={24} />
   300→   300→        </div>
   301→   301→        <div className="mr-auto">
   302→   302→          <h1 className="text-[26px] font-black leading-tight text-[var(--fg)]">Library Agent</h1>
   303→   303→          <div className="text-[12.5px] text-[var(--fg-dim)]">

   307→   307→        <Pill tone={cfg.enabled ? (runningNow ? "accent" : "good") : "neutral"}>
   308→   308→          {cfg.enabled ? (runningNow ? "Working…" : "Active") : "Paused"}
   309→   309→        </Pill>
   310→   310→        <button
   311→   311→          type="button"
   312→   312→          className="fs-pill"

   325→   325→          Scan library

   392→   392→                  <span>

   599→   599→function ScoreSparkline({ history }: { history: HistoryPoint[] }) {
   600→   600→  if (history.length === 0) return <p className="text-[13px] text-[var(--fg-dim)]">No history yet.</p>;
   601→   601→  const W = 100;
   602→   602→  const H = 32;
   603→   603→  const pts = history.map((h, i) => `${(i / Math.max(history.length - 1, 1)) * W},${H - (h.score / 100) * H}`).join(" ");
   604→   604→  const media = history.map((h, i) => `${(i / Math.max(history.length - 1, 1)) * W},${H - h.mediaReadable * H}`).join(" ");
   605→   605→  return (
   606→   606→    <div>
   607→   607→      <svg viewBox={`0 0 ${W} ${H}`} preserveAspectRatio="none" className="h-24 w-full" aria-label="Health score history chart">
   608→   608→        <polyline points={media} fill="none" stroke="rgba(255,255,255,0.18)" strokeWidth="0.8" vectorEffect="non-scaling-stroke" />
   609→   609→        <polyline points={pts} fill="none" stroke="var(--primary)" strokeWidth="1.4" vectorEffect="non-scaling-stroke" />
   610→   610→      </svg>
   611→   611→      <div className="mt-1 flex gap-4 text-[11px] text-[var(--fg-dim)]">
   612→   612→        <span className="flex items-center gap-1.5">
   613→   613→          <span className="inline-block h-0.5 w-4" style={{ background: "var(--primary)" }} /> health score
   614→   614→        </span>
   615→   615→        <span className="flex items-center gap-1.5">
   616→   616→          <span className="inline-block h-0.5 w-4 bg-white/20" /> media readability
   617→   617→        </span>

   619→   619→    </div>

   623→   623→function NumberField({ label, value, min, max, onCommit }: { label: string; value: number; min: number; max: number; onCommit: (v: number) => void }) {
   624→   624→  // draft state with adjust-during-render reset when the prop value changes
   625→   625→  const [state, setState] = useState({ lastProp: value, draft: String(value) });
   626→   626→  if (state.lastProp !== value) {
   627→   627→    setState({ lastProp: value, draft: String(value) });
   628→   628→  }
   629→   629→  return (
   630→   630→    <label className="flex flex-col gap-1.5">
   631→   631→      <span className="text-[11px] font-bold uppercase tracking-[0.08em] text-[var(--fg-dim)]">{label}</span>
   632→   632→      <input
   633→   633→        type="number"
   634→   634→        min={min}
   635→   635→        max={max}
   636→   636→        value={state.draft}
   637→   637→        onChange={(e) => setState({ ...state, draft: e.target.value })}
   638→   638→        onBlur={() => {
   639→   639→          const n = Number(state.draft);
   640→   640→          if (!Number.isNaN(n) && n >= min && n <= max && n !== value) onCommit(n);
   641→   641→          setState({ lastProp: value, draft: String(value) });
   642→   642→        }}
   643→   643→        className="fs-input h-9"
   644→   644→      />
   645→   645→    </label>
   646→   646→  );
   647→   647→}
